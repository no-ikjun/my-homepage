    if (typeof tag === "function") {
      return tag(props);
    }
    var element = { tag, props: { ...props, children } };
    console.log(element);
    return element;
  },
};

    <h1>hello</h1>
    <p>
      Lorem ipsum, dolor sit amet consectetur adipisicing elit. Ipsa harum perferendis veniam ut accusantium iusto deleniti nemo! Nisi quisquam natus itaque similique, adipisci voluptatum atque ex
      deleniti harum quibusdam inventore!
    </p>
  </div>
);

const render = (reactElement, container) => {
  const actualDomElement = document.createElement(reactElement.tag);
  if (reactElement.props) {
    Object.keys(reactElement.props)
      .filter((p) => p !== "children")
      .forEach((p) => (actualDomElement[p] = reactElement[p]));
  }
  if (reactElement.props.children) {
    reactElement.props.children.forEach((child) => render(child, actualDomElement));
  }
  container.appendChild(actualDomElement);
};

그래서 일단은 외부 스코프에 store을 만들어서 그곳에 값을 저장하는 방식을 구현해보자.

state를 보관 할 states와 어떤 state에 대해 다루는지를 알려줄 stateCursor를 useState를 선언한 부분 위에 각각 선언.

(stateCursor는 states의 몇 번째 인덱스에 그 값이 있는지를 나타낼 변수?)

```jsx
const states = [];
let stateCursor = 0;
```

일단 useState에 있는 콘솔 로그 부터 지우고,

forzenCursor라는 변수를 만들어 어떤 state에 대해서 다룰지 나타낸다.

원래 state라는 변수로 관리하던 값을 외부 변수인 states의 값을 참조하는 식으로 변경함.

initialState가 없는 경우에는 그냥 states[FROZENCURSOR]를 넣어줌

**states[FROZENCURSOR]에 값이 없을 경우에는 initialState를 넣어줌**

로직을 마무리할 때 stateCursor++를 해줌으로써 자동으로 다음 상태로 이동

```jsx
const states = [];
let stateCursor = 0;

const useState = (initialState) => {
  const FROZENCURSOR = stateCursor;
  states[FROZENCURSOR] = states[FROZENCURSOR] || initialState;
  const setState = (newState) => {
    states[FROZENCURSOR] = newState;
    rerender();
  };
  stateCursor++;
  return [states[FROZENCURSOR], setState];
};
```

그리고 컴포넌트가 리렌더링 되었을 때는 stateCursor를 0으로 초기화

```jsx
const rerender = () => {
  stateCursor = 0;
  document.getElementById("app").firstChild.remove();
  render(<App />, document.getElementById("app"));
};
```

createResource에 대한 부가 설명

- createResource()는 wrapPromise를 호출함.
- 여기서 promise의 상태에 접근할 수 없었던 문제를 해결함.
- 클로저로 상태와 결과를 관리하고 suspender를 throw하는 것으로 Suspense 컴포넌트에 promise를 전달할 수 있게 됨.
- pending상태일 경우에 상태에 접근할 수 있게 result를 throw하고,
- error상태일 경우에도 에러를 처리하기 위해 result를 throw함.

```tsx
const App = () => {
  const [name, setName] = useState("ikjun");
  const [count, setCount] = useState(0);
  const dogPhotoUrl = createResource(() => {
    fetch("https://dog.ceo/api/breeds/image/random")
      .then((r) => r.json())
      .then((playload) => playload.message), 'dogPhoto'
  });
  return (
    <div className="good">
      <h1>Hello, {name}</h1>
      <input type="text" placeholder="name" value={name} onchange={(e) => setName(e.target.value)} />
      <h2>The count is : {count}</h2>
      {dogPhotoUrl}
      <button onclick={() => setCount(count + 1)}>+</button>
      <button onclick={() => setCount(count - 1)}>-</button>
      <p>
        Lorem ipsum, dolor sit amet consectetur adipisicing elit. Ipsa harum perferendis veniam ut accusantium iusto deleniti nemo! Nisi quisquam natus itaque similique, adipisci voluptatum atque ex
        deleniti harum quibusdam inventore!
      </p>
    </div>
  );
};
```

그래서 일단 이런 식으로 수정. 

createResource는 데이터 불러오는 로직을 앞에 작성하고, 그 뒤에는 고유 key 값을 파라미터로 넣어주어야 한다.

그럼 이제 createResource를 만들어보자. (App 컴포넌트 바로 위에)

```tsx
const promiseCache = new Map();
const createResource = (asyncFunc, key) => {
  if(promiseCache.has(key)){
    return promiseCache.get(key);
  }
  throw { promise: asyncFunc(), key };
}
```

App 컴포넌트는 코드의 상단에 있는 createElement로 부터 만들어짐.

→ 자식 요소를 그냥 만들기만 하면 안됨.

→ 왜? promise 형태인 놈이 있기 때문!

따라서 try, catch문 사용

```tsx
const React = {
  createElement: (tag, props, ...children) => {
    if (typeof tag === "function") {
      try {
        return tag(props);
      } catch ({ promise, key }) {
        promise.then((data) => {
          promiseCache.set(key, data);
          rerender();
        });
        return { tag: "h1", props: { children: ["Loading..."] } };
      }
    }
    var element = { tag, props: { ...props, children } };
    return element;
  },
};
```

이런 식으로 promise가 있을 때의 상황을 관리해주면 됨.

아직 pending 상태이면 Loading이라는 문구를 출력

정상 처리된 상태이면 그냥 props 전달

마지막으로 불러온 데이터를 바탕으로 사진을 출력하도록 컴포넌트를 구성한다면
